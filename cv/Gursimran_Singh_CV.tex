\begin{entries}
  \entry{Doctoral Researcher, advised by Dr.\ H.\,B.\ Acharya}{2021 -- Present}
        {Oklahoma State University (2026 -- present) \,\textperiodcentered\, Rochester Institute of Technology (2021 -- 2025)}{}
  \vspace{1pt}
  \item \textbf{Programmable data-plane security on Intel Tofino} \hfill {\small 2021 -- Present}
  \begin{bullets}
    \item Implemented packet-inspection and stateful-filtering prototypes in P4 on Intel Tofino/Tofino2, working within parser,
          stage, memory, and instruction-set constraints.
    \item Developed and evaluated packet-processing logic using the P4 Studio SDE and BF-RT control plane.
    \item Investigate the relationship between completed inspection, enforcement timing, and throughput under constrained
          hardware resources.
    \item First-authored a systematization of programmable-data-plane security research, published at NeTCoM 2025 [P1].
  \end{bullets}
  \item \textbf{Reliable in-network computation} (ongoing research) \hfill {\small 2026 -- Present}
  \begin{bullets}
    \item Investigating which computation belongs in a programmable switch, on a SmartNIC/DPU, or on the host (CPU/GPU) for
          performance and robustness, and how in-network computations can recover from device failures while preserving correct
          results across that heterogeneous hardware, and at what cost.
  \end{bullets}
  \item \textbf{Internet measurement, censorship, and resilience} \hfill {\small 2022 -- Present}
  \begin{bullets}
    \item Designed and implemented measurement tooling for the Russia--Ukraine study: a classifier for \emph{soft} failures of
          website access and a multi-protocol traceroute used to investigate where filtering responses may originate.
    \item Collected and analyzed measurements from Ukraine, Russia, and the United States in March and July 2022 (ACM ASIA CCS 2023 [P2]).
    \item Ongoing: how reachability and paths change during disruption, and how measurement uncertainty and probe behaviour
          affect the conclusions.
  \end{bullets}
\end{entries}
